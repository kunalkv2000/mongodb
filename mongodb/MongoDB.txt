//Find all the topics and tasks which are thought in the month of October

db.createView("task_topics", "topics", [
   {
      $lookup: {
         from: "tasks",
         localField: "tp_id",
         foreignField: "tp_id",
         as: "tasksDocs"
      }
   },
   {
      $project: {
         _id: 0,
         name: 1,
         title: "$tasksDocs.title",
         Date: "$tasksDocs.Date" // Ensure this matches the actual field name
      }
   },
   {
      $unwind: "$title"
   },
   {
      $unwind: "$Date"
   },
   {
      $match: {
         Date: {
            $gte: ("2020-10-01"),
            $lte: ("2020-10-31")
         }
      }
   }
]);

db.createView("drive_placed", "company_drives", [
    {
        $lookup: {
            from: "users",         // Name of the users collection
            localField: "u_id", // Field in company_drives
            foreignField: "u_id",  // Field in users
            as: "usersDocs"        // Name of the output array
        }
    },
	
	{
      $project: {
         _id: 0,
         company_name: 1,
         name: "$usersDocs.name"
      }
   },
   
   {
      $unwind: "$name"
   }
])

//Find the number of users who are absent and task is not submitted between 15-oct-2020 and 31-oct-2020

db.createView("incomplete", "attendance", [
    {
        $match: {
            status: "Absent", // Filter absent users
            date: { 
                $gte: ("2020-10-15"), 
                $lte: ("2020-10-31") 
            }
        }
    },
    {
        $lookup: {
            from: "tasks", // Join with the tasks collection
            localField: "user_id", // Field from attendance to match
            foreignField: "assigned_to", // Field from tasks to match
            as: "taskDetails" // Name of the joined data
        }
    },
    {
        $match: {
            "taskDetails.status": { $in: ["InComplete"] } // Filter tasks not submitted
        }
    },
    {
        $group: {
            _id: null, // No specific grouping key, just count total users
            count: { $sum: 1 } // Count the number of records
        }
    }
]);
